def construir_ramp(brilho, calor):
    f = max(50.0, min(100.0, brilho)) / 100.0
    rf, gf, bf = 1 + .30*calor, 1 + .05*calor, 1 - .60*calor
    ramp = (ctypes.c_ushort * 768)()
    for i in range(256):
        k = i * 257
        ramp[i]     = min(65535, int(k*f*rf + .5))
        ramp[256+i] = min(65535, int(k*f*gf + .5))
        ramp[512+i] = min(65535, int(k*f*bf + .5))
    return ramp

class EMA:
    __slots__ = ("a", "v")
    def __init__(self, a, v=None): self.a, self.v = a, v
    def __call__(self, x):
        self.v = x if self.v is None else self.a*x + (1-self.a)*self.v
        return self.v

def fundir(pares):

    if total <= 0: return pares[0][0]

        melhor, ms = None, -1.0

class HoltWinters:
    def __init__(self, dados):
        self.nivel = dados.get("nivel") or {"b": None, "c": None, "e": None}
        self.tend = dados.get("tend") or {"b": 0.0, "c": 0.0, "e": 0.0}
        self.saz = {k: EMA(0.3, v) for k, v in (dados.get("sazon") or {}).items()}
    def observa(self, idx, b, c, e):
        for ch, x in (("b", b), ("c", c), ("e", e)):
            base = self.nivel[ch] if self.nivel[ch] is not None else x
            self.saz.setdefault(f"{ch}{idx}", EMA(0.3))(x - base)
            ant = self.nivel[ch]
            self.nivel[ch] = x if ant is None else .08*x + .92*ant
            if ant is not None:
                self.tend[ch] = .02*(self.nivel[ch]-ant) + .98*self.tend[ch]
    def preve(self, idx):
        r = {}
        for ch in "bce":
            if self.nivel[ch] is None: return None
            s = self.saz.get(f"{ch}{idx}")
            r[ch] = self.nivel[ch] + (s.v if s else 0.0) + self.tend[ch]
        return r
    def dump(self):
        return {"nivel": self.nivel, "tend": self.tend,
                "sazon": {k: e.v for k, e in self.saz.items() if e.v is not None}}

class MotorQuanticoSuave:
    def __init__(self, funcao_aplicar_gamma, trabalhador_ddc, log_fn=None,
                 passos_min=20, passos_max=100):
        self.aplicar_gamma = funcao_aplicar_gamma
        self.ddc = trabalhador_ddc
        self.log = log_fn or (lambda m: None)
        self.p_min, self.p_max = passos_min, passos_max
        self._lock = threading.Lock()
        self._token = 0
        self._transicao_real = False

    @property
    def transicao_em_curso(self):
        return self._transicao_real

    def transicionar(self, b_alvo, c_alvo, b_atual, c_atual):
        with self._lock:
            self._token += 1
            token = self._token
            self._transicao_real = True
        distancia = max(abs(b_alvo - b_atual), abs(c_alvo - c_atual))
        passos = max(self.p_min, min(self.p_max, int(distancia * 1.5)))
        intervalo = 0.025
        def _passos():
            for i in range(1, passos + 1):
                if token != self._token: return
                fracao = i / passos
                b_hora = b_atual + (b_alvo - b_atual) * fracao
                c_hora = c_atual + (c_alvo - c_atual) * fracao
                try:
                    self.aplicar_gamma(round(b_hora), round(c_hora))
                    if self.ddc and self.ddc.suporta:
                        self.ddc.fila.put(round(b_hora))
                except Exception as e:
                    self.log(f"[DRIVER] {e}")
                time.sleep(intervalo)
            self._transicao_real = False
            self.log(f"âœ¨ estabilizado: ({b_alvo}, {c_alvo}) em {passos} passos")
        threading.Thread(target=_passos, daemon=True).start()

def nascer_poser(lat):
    n = time.localtime().tm_yday
    lat = max(-65.0, min(65.0, float(lat)))

    x = max(-1.0, min(1.0, -math.tan(math.radians(lat)) * math.tan(math.radians(decl))))

    def _alvo_fundido(self):
        h = time.localtime().tm_hour
        idx = time.localtime().tm_wday*24 + h
        hab = self.cfg["aria"]["habito"].get(str(h))
        hw = self.hw.preve(idx)
        fontes, pares = [], []
        if hab:
            pares.append((hab, 25)); fontes.append(("hÃ¡bito", hab[0], 25))
        if hw:
            pares.append(([hw["b"], hw["c"], hw["e"]], 36))
            fontes.append(("semana", hw["b"], 36))
        if self._bandit_atual:
            p = CAND[self._bandit_atual]

        if not pares: return NOITE if self._eh_noite() else DIA, fontes

                     for i in range(3))

def _norm(texto):
    texto = unicodedata.normalize("NFKC", texto).lower()
    texto = "".join(c for c in unicodedata.normalize("NFD", texto) if unicodedata.category(c) != "Mn")
    return texto

def _match(texto_norm, keywords):
    return any(_norm(k) in texto_norm for k in keywords)

def detectar_intencao(prompt):
    p = _norm(prompt)
    melhores = []
    for categoria, intents in ONTOLOGIA.items():
        for nome, config in intents.items():
            score = sum(1 for kw in config.get("keywords", []) if _norm(kw) in p)
            if score > 0:
                melhores.append({"categoria": categoria, "intencao": nome, "score": score})
    if not melhores:
        return None
    melhores.sort(key=lambda x: -x["score"])
    return melhores[0]

def detectar_features(prompt):
    p = _norm(prompt)
    return [nome for nome, cfg in FEATURES.items() if _match(p, cfg["keywords"])]



--- FUNCAO ---
def detectar_posicoes(prompt):
    p = _norm(prompt)
    return [nome for nome, cfg in POSICOES.items() if _match(p, cfg["keywords"])]

def detectar_modificadores(prompt):
    p = _norm(prompt)
    mods = {"tamanho": None, "estilo": None}
    for t, kws in TAMANHOS.items():
        if _match(p, kws): mods["tamanho"] = t; break
    for e, kws in ESTILOS.items():
        if _match(p, kws): mods["estilo"] = e; break
    return mods

def compor_chave(prompt):
    intencao = detectar_intencao(prompt)
    if not intencao:
        return {
            "chave": "desconhecido",
            "intencao": None,
            "categoria": None,
            "features": [],
            "posicoes": [],
            "modificadores": {},
        }
    features = sorted(detectar_features(prompt))
    posicoes = sorted(detectar_posicoes(prompt))
    mods = detectar_modificadores(prompt)
    
    partes = [intencao["intencao"]]
    if features: partes.append("_".join(features))
    if posicoes: partes.append("_".join(posicoes))
    if mods.get("tamanho"): partes.append(mods["tamanho"])
    if mods.get("estilo"): partes.append(mods["estilo"])
    
    return {
        "chave": "__".join(partes),
        "intencao": intencao["intencao"],
        "categoria": intencao["categoria"],
        "features": features,
        "posicoes": posicoes,
        "modificadores": mods,
    }

    words = texto.lower().split()

def similaridade_cosseno(v1, v2):
    if not v1 or not v2: return 0.0
    intersecao = set(v1).intersection(set(v2))
    return len(intersecao) / math.sqrt(len(v1) * len(v2))

def buscar_no_genoma(prompt, memoria):
    """
    Le o prompt, vetoriza, e compara com todas as chaves do dynamic_tools.json
    para achar a correspondencia semantica mais proxima.
    """
    if not memoria:
        return None, 0.0
        
    vetor_prompt = obter_vetor(prompt)
    melhor_chave = None
    melhor_score = 0.0

    for chave, entry in memoria.items():
        # Compara o prompt com a chave (ex: "jogo_da_velha")
        # e com as features (ex: ["som", "cor"])
        texto_alvo = chave + " " + " ".join(entry.get("features", [])) if isinstance(entry, dict) else chave
        vetor_alvo = obter_vetor(texto_alvo)
        
        score = similaridade_cosseno(vetor_prompt, vetor_alvo)
        if score > melhor_score:
            melhor_score = score
            melhor_chave = chave

    return melhor_chave, melhor_score

def _bater_chave(chave, prompt):
    try:
        if chave.startswith("sk-or-v1"):
            url = "https://openrouter.ai/api/v1/chat/completions"
            headers = {"Authorization": f"Bearer {chave}", "Content-Type": "application/json"}
            payload = {"model": "deepseek/deepseek-chat", "messages": [{"role": "user", "content": prompt}]}
            provider = "OpenRouter"
        elif chave.startswith("gsk_"):
            url = "https://api.groq.com/openai/v1/chat/completions"
            headers = {"Authorization": f"Bearer {chave}", "Content-Type": "application/json"}
            payload = {"model": "llama3-8b-8192", "messages": [{"role": "user", "content": prompt}]}
            provider = "Groq"
        elif chave.startswith("sk-") and not chave.startswith("sk-or-v1"):
            url = "https://api.deepseek.com/chat/completions"
            headers = {"Authorization": f"Bearer {chave}", "Content-Type": "application/json"}
            payload = {"model": "deepseek-chat", "messages": [{"role": "user", "content": prompt}]}
            provider = "DeepSeek"
        elif chave.startswith("AIza") or chave.startswith("AQ."):
            url = f"https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key={chave}"
            headers = {"Content-Type": "application/json"}
            payload = {"contents": [{"parts": [{"text": prompt}]}]}
            provider = "Gemini"
        else:
            return (chave, None, "Desconhecido")

        req = urllib.request.Request(url, data=json.dumps(payload).encode("utf-8"), headers=headers, method="POST")
        with urllib.request.urlopen(req, timeout=12) as response:
            res_body = response.read().decode("utf-8", errors="replace")
            res_json = json.loads(res_body)
            if provider == "Gemini": texto = res_json["candidates"][0]["content"]["parts"][0]["text"]
            else: texto = res_json["choices"][0]["message"]["content"]
            return (chave, texto, None)
    except urllib.error.HTTPError as e:
        return (chave, None, f"HTTP {e.code}")
    except Exception as e:
        return (chave, None, str(e))

def disparar_enxame(prompt, anel):
    if not anel.keys: return None, "KeyRing vazio"
    chaves_ativas = [k for k in anel.keys if k not in anel.dead_keys]
    if not chaves_ativas: return None, "Todas em cooldown"

    print(f"[ENXAME] Disparando {len(chaves_ativas)} chaves em paralelo...")
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=len(chaves_ativas)) as executor:
        futuros = {executor.submit(_bater_chave, k, prompt): k for k in chaves_ativas}
        done, _ = concurrent.futures.wait(futuros, timeout=15)
        
        for futuro in done:
            chave, resposta, erro = futuro.result()
            if resposta:
                print(f"[ENXAME] Vitoria! O Enxame respondeu.")
                executor.shutdown(wait=False, cancel_futures=True)
                return resposta, "Enxame Nuvem"
            elif erro in ["HTTP 402", "HTTP 429", "HTTP 401"]:
                anel.mark_dead(chave, cooldown_hours=1)
                
    print("[ENXAME] Todas as chaves falharam ou deram timeout.")
    return None, "Falha Total"
